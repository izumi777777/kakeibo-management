💡 Databricks での活用例

ダウンロードした CSV を Databricks にアップロードして分析できます。 

# Databricks Notebook サンプル
df = spark.read.csv(
    "/FileStore/kakeibo/transactions_*.csv",
    header=True,
    inferSchema=True,
    encoding="UTF-8"
)

# 月別支出集計
from pyspark.sql.functions import month, year, sum as _sum
df.filter(df.is_income == "FALSE") \
  .groupBy(year("transaction_date"), month("transaction_date"), "transaction_type") \
  .agg(_sum("amount").alias("total")) \
  .orderBy("year(transaction_date)", "month(transaction_date)") \
  .show()
